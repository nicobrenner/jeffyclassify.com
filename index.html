Predict

from jeffy.engine import Engine engine = Engine() engine.load() result = engine.predict( "banking77", "I was charged twice for the same transaction" ) # {"label": "transaction_charged_twice", "confidence": 0.999}

Train your own

from jeffy.train import train_classifier clf = train_classifier( texts=["great product!", "terrible", ...], labels=["positive", "negative", ...], task_id="my_reviews", ) clf.predict("exceeded expectations") # {"label": "positive", "confidence": 0.87}
